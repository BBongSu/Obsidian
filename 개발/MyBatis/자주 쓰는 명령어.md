``` xml title="특정 문자열 데이터 자동기입"
<selectKey keyProperty="ppmSndCode" resultType="string" order="BEFORE">
	SELECT
		CONCAT('SMS-', LPAD( IFNULL(MAX(CAST(SUBSTRING(ppmSndCode, 5) AS UNSIGNED)), 0) + 1, 4, '0'))
	FROM bpp_por_msg
	WHERE ppmSndCode LIKE 'SMS-%'
</selectKey>
```
