		CONCAT('SMS-', LPAD( IFNULL(MAX(CAST(SUBSTRING(ppmSndCode, 5) AS UNSIGNED)), 0) + 1, 4, '0'))
	FROM bpp_por_msg
	WHERE ppmSndCode LIKE 'SMS-%'
