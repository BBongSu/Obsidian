``` kotlin title="타 앱으로 Parameter 전송 (Kotlin WebViewScreen.kt)"
package com.example.bizpack.ui.fragment  
  
import android.annotation.SuppressLint  
import android.net.Uri  
import android.webkit.CookieManager  
import android.webkit.ValueCallback  
import android.webkit.WebChromeClient  
import android.webkit.WebSettings  
import android.webkit.WebView  
import android.webkit.WebViewClient  
import androidx.activity.compose.rememberLauncherForActivityResult  
import androidx.activity.result.contract.ActivityResultContracts  
import androidx.compose.foundation.layout.fillMaxSize  
import androidx.compose.runtime.Composable  
import androidx.compose.runtime.mutableStateOf  
import androidx.compose.runtime.remember  
import androidx.compose.ui.Modifier  
import androidx.compose.ui.viewinterop.AndroidView  
import com.example.bizpack.ui.viewmodel.WebAppInterface  
  
@SuppressLint("SetJavaScriptEnabled")  
@Composable  
fun WebViewScreen(url: String, location: String) {  
    // 파일 선택 콜백 관리  
    val filePathCallbackState = remember { mutableStateOf<ValueCallback<Array<Uri>>?>(null) }  
  
    // 파일 선택을 처리하는 런처  
    val fileChooserLauncher = rememberLauncherForActivityResult(ActivityResultContracts.GetContent()) { uri ->  
        filePathCallbackState.value?.onReceiveValue(uri?.let { arrayOf(it) })  
        filePathCallbackState.value = null  
    }  
  
    AndroidView(  
        modifier = Modifier.fillMaxSize(),  
        factory = { context ->  
            WebView(context).apply {  
                settings.javaScriptEnabled = true  
                settings.domStorageEnabled = true  
                settings.allowFileAccess = true  
                settings.cacheMode = WebSettings.LOAD_NO_CACHE  
                settings.useWideViewPort = true  
                settings.loadWithOverviewMode = true  
  
                // 캐시 삭제  
                clearCache(true)  
  
                val cookieManager = CookieManager.getInstance()  
                cookieManager.setAcceptCookie(true)  
                cookieManager.setAcceptThirdPartyCookies(this, true)  
  
                // WebAppInterface 등록  
                addJavascriptInterface(WebAppInterface(context), "Android")  
  
                // WebViewClient 설정  
                webViewClient = WebViewClient()  
  
                // WebChromeClient 설정  
                webChromeClient = object : WebChromeClient() {  
                    override fun onShowFileChooser(  
                        webView: WebView?,  
                        filePathCallback: ValueCallback<Array<Uri>>?,  
                        fileChooserParams: FileChooserParams?  
                    ): Boolean {  
                        // 파일 선택 콜백 저장  
                        filePathCallbackState.value = filePathCallback  
                        fileChooserLauncher.launch("image/*") // 이미지 파일만 선택 가능  
                        return true  
                    }  
                }  
  
                // URL 로드  
                loadUrl(url)  
            }  
        }    )  
}
```

``` kotlin title="타 앱으로 Parameter 전송 (Kotlin WebAppInterface.kt)"
package com.example.bizpack.ui.viewmodel  
  
import android.content.Context  
import android.content.Intent  
import android.net.Uri  
import android.util.Log  
import android.webkit.JavascriptInterface  
import android.widget.Toast  
import com.example.bizpack.util.FileUtils  
  
  
class WebAppInterface(private val context: Context) {  
  
    // JavaScript에서 호출 가능한 함수  
    @JavascriptInterface  
    fun saveSignature(base64Image: String) {  
        FileUtils.saveSignatureToGallery(context, base64Image)  
    }  
  
    @JavascriptInterface  
    fun requestPayment(price: String, catid: String){  
        try {  
            val returnURI = "testapp://result"  
            val uriStr = "apppos://pay?gubun=0100&bizno=2208115770&price=$price&catid=$catid&returnURI=$returnURI"  
  
            Log.d("APP_POS", "결제 URI : $uriStr")  
  
            val intent = Intent(Intent.ACTION_VIEW, Uri.parse(uriStr))  
            if (intent.resolveActivity(context.packageManager) != null) {  
                context.startActivity(intent)  
            } else {  
                Toast.makeText(context, "APP POS 앱이 설치되어 있지 않습니다.", Toast.LENGTH_SHORT).show()  
            }  
        } catch (e: Exception) {  
            e.printStackTrace()  
            Toast.makeText(context, "결제 요청 중 오류 발생", Toast.LENGTH_SHORT).show()  
        }  
    }  
}
```
