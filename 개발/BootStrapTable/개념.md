``` javascript title="테이블 row 삭제"
/*

1. 하드코딩된 테이블 ID나 인덱스(`listIdx`)에 의존하지 않고, 클릭된 버튼(`btn`) 객체로부터 DOM 트리를 타고 올라가 가장 가까운 테이블 엘리먼트를 찾도록 개선

2. 문자형 키와 숫자형 키 모두 유연하게 대응할 수 있도록 `values` 매칭 배열에 문자열형, 숫자형 값을 모두 포함시켜 매칭되도록 처리

*/
window.fnc_tblDelProductRow = function(btn, irSeq, listIdx) {
	Swal.fire({
		title: '삭제하시겠습니까?',
		text: "이 항목은 복구할 수 없습니다.",
		icon: 'warning',
		showCancelButton: true,
		confirmButtonText: '삭제',
		cancelButtonText: '취소'
	}).then((result) => {
		if (result.isConfirmed) {
			const $table = $(btn).closest('table');
				$table.bootstrapTable('remove', {
					field: 'irSeq',
					values: [irSeq, Number(irSeq), String(irSeq)]
				});
				fnc_calcTotAmt($table);
		}
	});
};
```
