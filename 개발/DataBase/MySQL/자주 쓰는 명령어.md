``` sql title="테이블 10만개 더미 데이터 테스트"
DROP PROCEDURE IF EXISTS bizpack_dev.dummy_test;

CREATE PROCEDURE bizpack_dev.dummy_test()
BEGIN
	DECLARE i INT DEFAULT 1;
	DECLARE bulk_size INT DEFAULT 1000;
	DECLARE sql_str LONGTEXT DEFAULT '';
	DECLARE counter INT DEFAULT 0;

SET sql_str = 'INSERT INTO toast_test(id, name, artist, TYPE, releas, genre) VALUES ';

WHILE i <= 100000 DO
	SET sql_str = CONCAT(sql_str, '(', i, ', ', '''', CONCAT('name', i), '''', ', ', '''', CONCAT('artist', i), '''', ', ', '''', CONCAT('type', i), '''', ', ', '''', CONCAT('releas', i), '''', ', ', '''', CONCAT('genre', i), '''', ')');
	SET counter = counter + 1;
	SET i = i + 1;

	-- Insert every bulk_size rows
	IF counter = bulk_size THEN
		SET sql_str = CONCAT(sql_str, ';');
		PREPARE stmt FROM sql_str;
		EXECUTE stmt;
		DEALLOCATE PREPARE stmt;

	-- Reset for next bulk
	SET sql_str = 'INSERT INTO toast_test(id, name, artist, TYPE, releas, genre) VALUES ';
	SET counter = 0;
	ELSE
		-- Add a comma if not the last row in the bulk
		IF i <= 100000 THEN
			SET sql_str = CONCAT(sql_str, ', ');
		END IF;
	END IF;
END WHILE;

-- Insert remaining rows if any
	IF counter > 0 THEN
		SET sql_str = CONCAT(sql_str, ';');
		PREPARE stmt FROM sql_str;
		EXECUTE stmt;
		DEALLOCATE PREPARE stmt;
	END IF;
END;

CALL dummy_test();
```

``` sql title="전체 테이블 컬럼 추가 (테이블, 컬럼 네이밍 규칙이 존재할 시)"
CREATE DEFINER=`bizpack_dev`@`175.193.237.63` PROCEDURE `bizpack_dev_test`.`fnCreateColumnDocNo`()

BEGIN

DECLARE done INT DEFAULT FALSE;

DECLARE t_name VARCHAR(255);

DECLARE col_name VARCHAR(100);

DECLARE sql_query TEXT;

-- 1. 대상 테이블 선정 (예: 특정 접두어로 시작하는 테이블들)

DECLARE cur CURSOR FOR

SELECT TABLE_NAME

FROM information_schema.TABLES

WHERE TABLE_SCHEMA = 'table_name'

AND TABLE_NAME LIKE 'bpp_%'; -- 규칙에 맞는 테이블만 필터링

DECLARE CONTINUE HANDLER FOR NOT FOUND SET done = TRUE;

  

OPEN cur;

  

read_loop: LOOP

FETCH cur INTO t_name;

IF done THEN

LEAVE read_loop;

END IF;

  

-- 2. 컬럼명 생성 로직 (예: bpp_por_user_log -> ppulDocNo)

-- 첫 번째 단어의 마지막 글자 (직관적으로 뒤집어서 첫 글자 따기)

-- 두 번째 단어부터 마지막 단어까지의 '첫 글자'들만 추출

SET col_name = (

SELECT CONCAT(

SUBSTR(SUBSTRING_INDEX(t_name, '_', 1), -1),

REGEXP_REPLACE(

SUBSTR(t_name, INSTR(t_name, '_') + 1), -- 첫 '_' 이후 문자열

'([A-Za-z])[A-Za-z]*(_|$)', -- 단어의 첫 글자만 캡처하고 나머지는 무시

'\\1' -- 캡처한 첫 글자만 남김

),

'DocNo'

/*SUBSTRING(SUBSTRING_INDEX(t_name, '_', 1), -1), -- 1번째 단어 끝 (p)

SUBSTRING(SUBSTRING_INDEX(t_name, '_', 2), 1, 1), -- 2번째 단어 시작 (p)

SUBSTRING(SUBSTRING_INDEX(SUBSTRING_INDEX(t_name, '_', 3), '_', -1), 1, 1), -- 3번째 시작 (u)

SUBSTRING(SUBSTRING_INDEX(SUBSTRING_INDEX(t_name, '_', 4), '_', -1), 1, 1), -- 4번째 시작 (l)

'DocNo'*/

)

);

  

-- 3. 동적 SQL 생성 및 실행

SET sql_query = CONCAT('ALTER TABLE ', t_name, ' ADD COLUMN ', col_name, ' VARCHAR(50) DEFAULT NULL COMMENT "[기본정보] 문서번호"');

-- 이미 컬럼이 있는지 체크 후 실행 (에러 방지)

IF NOT EXISTS (

SELECT * FROM information_schema.COLUMNS

WHERE TABLE_SCHEMA = 'table_name'

AND TABLE_NAME = t_name

AND COLUMN_NAME = col_name

) THEN

SET @v_sql = sql_query;

PREPARE stmt FROM @v_sql;

EXECUTE stmt;

DEALLOCATE PREPARE stmt;

END IF;

END LOOP;

  

CLOSE cur;

END
```
