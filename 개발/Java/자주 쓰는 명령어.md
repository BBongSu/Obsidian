``` java title="[Java] API DTO <=> DB 컬럼으로 변경"
import java.io.IOException;
import java.util.Arrays;

public class Main {
	public static void main(String[] args) throws IOException {
		Class<?> clazz = com.baroservice.ws.TaxInvoice.class;
		
		Arrays
		.stream(clazz.getDeclaredMethods())
		.filter(m -> m.getName().startsWith("set") && m.getParameterCount() == 1).sorted((a, b) -> a.getName().compareToIgnoreCase(b.getName())) // 선언 순서대로 정렬
		.forEach(method -> {
			String name = method.getName().substring(3); // set 제거
			System.out.println(name);
		});
	}
}
```

``` java title="[Java] DB 테이블, 컬럼, 설명 조회"
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.ResultSet;
import java.sql.Statement;

public class InspectDb {
    public static void main(String[] args) {
        String url = "jdbc:mariadb://59.6.178.99:33123/bizsys_dev?useUnicode=true&serverTimezone=UTC&characterEncoding=utf8";
        String user = "bizsys_dev";
        String pass = "bizsys_dev@Pass";
        
        try (Connection conn = DriverManager.getConnection(url, user, pass);
             Statement stmt = conn.createStatement()) {
            
            System.out.println("=== Connected to DB successfully ===");
            String query = "SELECT TABLE_NAME, COLUMN_NAME, DATA_TYPE, COLUMN_COMMENT " +
                           "FROM INFORMATION_SCHEMA.COLUMNS " +
                           "WHERE TABLE_SCHEMA = 'bizsys_dev' AND (TABLE_NAME LIKE '%sal_rcv%') " +
                           "ORDER BY TABLE_NAME, ORDINAL_POSITION";
                           
            ResultSet rs = stmt.executeQuery(query);
            String currentTable = "";
            while (rs.next()) {
                String table = rs.getString("TABLE_NAME");
                if (!table.equals(currentTable)) {
                    currentTable = table;
                    System.out.println("\n[TABLE: " + currentTable + "]");
                }
                System.out.printf("  %-30s : %-15s // %s%n", 
                    rs.getString("COLUMN_NAME"), 
                    rs.getString("DATA_TYPE"), 
                    rs.getString("COLUMN_COMMENT") != null ? rs.getString("COLUMN_COMMENT") : "");
            }
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
```
