``` java title="내장 Class의 setter getter 함수 전체로드"
import java.io.IOException;
import java.lang.reflect.Field;

public class Main {
	public static void main(String[] args) throws IOException {
		for (Field field : [Class명].class.getDeclaredFields()) {
			String name = field.getName();
			String method = "set" + Character.toUpperCase(name.charAt(0)) + name.substring(1);
			System.out.println("[Class 변수명]." + method + "(\"\");");
		}
	}
}
```

``` java title="API DTO <=> DB 컬럼으로 변경"
import java.io.IOException;
import java.util.Arrays;

public class Main {
	public static void main(String[] args) throws IOException {
		Class<?> clazz = com.baroservice.ws.TaxInvoice.class;
		
		Arrays
		.stream(clazz.getDeclaredMethods())
		.filter(m -> m.getName().startsWith("set") && m.getParameterCount() == 1).sorted((a, b) -> a.getName().compareToIgnoreCase(b.getName())) // 선언 순서대로 정렬
		.forEach(method -> {
			String name = method.getName().substring(3); // set 제거
			System.out.println(name);
		});
	}
}
```
