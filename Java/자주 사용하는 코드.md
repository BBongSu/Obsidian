``` java title="BufferedReader (백준)"
import java.io.BufferedReader;
import java.io.IOException;
import java.io.InputStreamReader;

public class Main {
	public static void main(String[] args) throws IOException {
		BufferedReader br = new BufferedReader(new InputStreamReader(System.in));
		StringBuilder sb = new StringBuilder();
		String[] str = br.readLine().split(" ");

		int basketCount, changeCount;
		int i,j,temp=0;

		basketCount = Integer.parseInt(str[0]);
		changeCount = Integer.parseInt(str[1]);
	
		if( (basketCount >= 1 && basketCount <= 100) && (changeCount >= 1 && basketCount <= 100) ) {
			int[] basketArray = new int[basketCount];
			for (int c = 0; c < basketArray.length; c++) {
				basketArray[c] = c+1;
			}
			for (int a = 0; a < changeCount; a++) {
				String[] str2 = br.readLine().split(" ");
				i = Integer.parseInt(str2[0]);
				j = Integer.parseInt(str2[1]);
				
				if( (i >= 1 && i <= j) && (j <= basketCount) ) {
					temp = basketArray[i-1];
					basketArray[i-1] = basketArray[j-1];
					basketArray[j-1] = temp;
				}
			}
	
			for (int k : basketArray) {
				sb.append(k).append(" ");
			}
			System.out.print(sb);
		}
	
		br.close();
	}
}
```

``` java title="DFS - 부분집합 구하기 (백준)"
public class Main {
	static int n;
	static boolean[] result;
	static StringBuilder sb;
	
	public static void main(String[] args) {
		n = 3;
		result = new boolean[n];
		dfs(0);
	}
	
	static void dfs(int k) {
		if(k == n) {
			sb = new StringBuilder();
			for(int i = 0; i < n; i++) {
				if(result[i]) sb.append(i+1);
			}
			if(sb.length() > 0) System.out.println(sb.toString());
			return;
		}

		result[k] = true;
		dfs(k+1);
		result[k] = false;
		dfs(k+1);
	}
}
/*

	부분 집합 구하기 (DFS)

	A 집합 원소가 n개라고 가정했을 때 A 집합의 부분집합 총 개수는 2^n - 1개 (공집합 제외)

	ex)

	1부터 3까지 자연수들을 원소로 가지는 집합 구하기..

	k=0 true -> k=1 true -> k=2 true => 123
	
						 -> k=2 false => 12
						 
			 -> k=1 false -> k=2 true => 13
			 
						  -> k=2 false => 1

	k=0 false -> k=1 true -> k=2 true => 23

						  -> k=2 false => 2

			  -> k=1 false -> k=2 true => 3
			  
						   -> k=2 false => 공집합
```
